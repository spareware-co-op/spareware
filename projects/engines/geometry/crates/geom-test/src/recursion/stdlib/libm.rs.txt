// The recursion check's model of the `libm` crate (MIT, https://github.com/rust-lang/libm),
// read in place of its source (src/recursion/stdlib.rs): its functions' signatures, from its
// documentation (https://docs.rs/libm/0.2.16/libm/). Every one computes on numbers it is
// handed and calls nothing back, so every body is `loop {}`. Never compiled.

macro_rules! unary {
    ($($f:ident $g:ident)*) => {
        $(
            pub fn $f(x: f64) -> f64 {
                loop {}
            }
            pub fn $g(x: f32) -> f32 {
                loop {}
            }
        )*
    };
}

    acos acosf acosh acoshf asin asinf asinh asinhf atan atanf atanh atanhf cbrt cbrtf ceil ceilf
    cos cosf cosh coshf erf erff erfc erfcf exp expf exp10 exp10f exp2 exp2f expm1 expm1f fabs fabsf
    floor floorf j0 j0f j1 j1f lgamma lgammaf log logf log10 log10f log1p log1pf log2 log2f rint rintf
    round roundf roundeven roundevenf sin sinf sinh sinhf sqrt sqrtf tan tanf tanh tanhf tgamma tgammaf
    trunc truncf y0 y0f y1 y1f
}

                loop {}
            }
            pub fn $g(x: f32, y: f32) -> f32 {
                loop {}
            }
        )*
    };
}
